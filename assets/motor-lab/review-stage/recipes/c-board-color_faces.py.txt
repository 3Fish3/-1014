def material_for(label,face):
 b=face.BoundBox
 # Existing shallow engraved/raised lid markings; keep the large surrounding lid dark.
 if label=='2_0' and b.ZMin>=-6.05 and b.ZMax<=-5.99 and face.Area<200:
  return 4

 dx,dy,dz=b.XLength,b.YLength,b.ZLength

 # Chip package and large central component, not connector plastic.
